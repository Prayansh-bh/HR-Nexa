function DocumentsPage(p: PageProps & { sub: string }) {
  const { sub, docs, setDocs, employees, onViewDoc, toast, user, empDocs, setEmpDocs } = p;

  if (sub === "documents.employee") return (
    <EmployeeDocsPage
      empDocs={empDocs} setEmpDocs={setEmpDocs}
      employees={employees}
      canEdit={perm(user.role, "documents.employee").write}
      scopeToSelf={!!perm(user.role, "documents.employee").scopeToSelf}
      userName={user.name}
      toast={toast}
    />
  );

  const titleMap: Record<string, { title: string; type: DocRecord["type"] }> = {
    "documents.offer": { title: "Offer Letters", type: "Offer" },
    "documents.appointment": { title: "Appointment Letters", type: "Appointment" },
    "documents.experience": { title: "Experience Letters", type: "Experience" },
  };
  const { title, type } = titleMap[sub] || { title: "Documents", type: "Offer" as const };
  const list = docs.filter(d => d.type === type);
  const canEdit = perm(user.role, sub).write;

  const [open, setOpen] = useState(false);
  const [empName, setEmpName] = useState(employees[0]?.name || "");

  const generate = () => {
    const emp = employees.find(e => e.name === empName);
    if (!emp) return;
    const id = `DOC${String(docs.length + 1).padStart(3, "0")}`;
    setDocs(prev => [{ id, employee: emp.name, type, issued: new Date().toISOString().slice(0,10), salary: emp.salary, designation: emp.designation }, ...prev]);
    toast(`${type} letter generated for ${emp.name}`);
    setOpen(false);
  };

  return (
    <div>
      <PageHeader title={title} subtitle="Generate and manage employee documents."
        action={canEdit && <Btn onClick={() => setOpen(true)}><Plus className="h-4 w-4" /> Generate</Btn>} />
      <div className="grid grid-cols-1 gap-4 md:grid-cols-2 lg:grid-cols-3">
        {list.map(d => (
          <Card key={d.id}>
            <div className="flex items-start gap-3">
              <div className="grid h-11 w-11 place-items-center rounded-xl btn-gradient"><FileText className="h-5 w-5" /></div>
              <div className="min-w-0 flex-1">
                <div className="truncate font-bold">{d.employee}</div>
                <div className="text-xs text-muted-foreground">Issued {d.issued}</div>
              </div>
            </div>
            <div className="mt-4 grid grid-cols-2 gap-2">
              <Btn variant="outline" size="sm" onClick={() => onViewDoc(d)}><Eye className="h-4 w-4" /> View</Btn>
              <Btn variant="outline" size="sm" onClick={() => generateDocumentPDF(d, p.company || "Acme Corp")}><Download className="h-4 w-4" /> Download</Btn>
            </div>
          </Card>
        ))}
        {list.length === 0 && (
          <Card className="md:col-span-2 lg:col-span-3 text-center py-12 text-sm text-muted-foreground">No {type.toLowerCase()} letters yet.</Card>
        )}
      </div>
      {open && (
        <ModalShell title={`Generate ${type} Letter`} onClose={() => setOpen(false)}>
          <div className="space-y-4">
            <Field label="Employee">
              <select value={empName} onChange={e => setEmpName(e.target.value)} className={inputCls}>
                {employees.map(e => <option key={e.id}>{e.name}</option>)}
              </select>
            </Field>
            <div className="rounded-lg bg-muted/40 p-3 text-xs text-muted-foreground">
              This will create a merged letter using the employee's designation, salary, and join date.
            </div>
          </div>
          <div className="mt-6 flex justify-end gap-2"><Btn variant="outline" onClick={() => setOpen(false)}>Cancel</Btn><Btn onClick={generate}>Generate</Btn></div>
        </ModalShell>
      )}
    </div>
  );
}

function EmployeeDocsPage({ empDocs, setEmpDocs, employees, canEdit, scopeToSelf, userName, toast }: {
  empDocs: EmpDoc[]; setEmpDocs: (d: EmpDoc[] | ((p: EmpDoc[]) => EmpDoc[])) => void;
  employees: Employee[]; canEdit: boolean; scopeToSelf: boolean; userName: string; toast: (m: string) => void;
}) {
  const [open, setOpen] = useState(false);
  const [selectedEmp, setSelectedEmp] = useState(scopeToSelf ? userName : (employees[0]?.name || ""));
  const [docType, setDocType] = useState("Aadhaar Card");
  const [fileName, setFileName] = useState("");
  const [fileObj, setFileObj] = useState<{ name: string; size: string; dataUrl?: string } | null>(null);
  const visibleDocs = scopeToSelf ? empDocs.filter(d => d.employee === userName) : empDocs;

  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (file) {
      const reader = new FileReader();
      reader.onload = () => {
        setFileObj({
          name: file.name,
          size: `${Math.round(file.size / 1024)} KB`,
          dataUrl: reader.result as string
        });
        setFileName(file.name);
      };
      reader.readAsDataURL(file);
    }
  };

  const upload = () => {
    if (!docType.trim()) return;
    const id = `ED${String(empDocs.length + 1).padStart(3, "0")}`;
    const name = fileName || (fileObj ? fileObj.name : `${docType.toLowerCase().replace(/\s+/g,"_")}_${selectedEmp.split(" ")[0].toLowerCase()}.pdf`);
    const size = fileObj ? fileObj.size : "120 KB";
    const fileData = fileObj?.dataUrl || "";
    setEmpDocs(prev => [{ id, employee: selectedEmp, docType, fileName: name, uploaded: new Date().toISOString().slice(0,10), size, fileData }, ...prev]);
    toast(`Document uploaded for ${selectedEmp}`);
    setOpen(false); setDocType("Aadhaar Card"); setFileName(""); setFileObj(null);
  };

  return (
    <div>
      <PageHeader
        title={scopeToSelf ? "My Documents" : "Employee Documents"}
        subtitle={scopeToSelf ? "Your personal documents stored securely." : "All employee personal documents."}
        action={canEdit && <Btn onClick={() => setOpen(true)}><Upload className="h-4 w-4" /> Upload Document</Btn>}
      />
      <Card className="!p-0 overflow-hidden">
        <div className="overflow-x-auto">
          <table className="w-full text-sm">
            <thead><tr className="border-b border-border bg-muted/40 text-left text-xs uppercase text-muted-foreground">
              {!scopeToSelf && <th className="px-4 py-3">Employee</th>}
              <th className="px-4 py-3">Document Type</th>
              <th className="px-4 py-3">File Name</th>
              <th className="px-4 py-3">Uploaded</th>
              <th className="px-4 py-3">Size</th>
              <th className="px-4 py-3 text-right">Actions</th>
            </tr></thead>
            <tbody>
              {visibleDocs.map(d => (
                <tr key={d.id} className="border-b border-border/60">
                  {!scopeToSelf && <td className="px-4 py-3 font-semibold">{d.employee}</td>}
                  <td className="px-4 py-3"><Badge tone="indigo">{d.docType}</Badge></td>
                  <td className="px-4 py-3 text-muted-foreground">{d.fileName}</td>
                  <td className="px-4 py-3">{d.uploaded}</td>
                  <td className="px-4 py-3">{d.size}</td>
                  <td className="px-4 py-3 text-right">
                    <button 
                      onClick={() => {
                        if (d.fileData) {
                          const link = document.createElement("a");
                          link.href = d.fileData;
                          link.download = d.fileName;
                          link.click();
                        } else {
                          const blob = new Blob(["Simulated Document content for " + d.fileName], { type: "application/pdf" });
                          const url = URL.createObjectURL(blob);
                          const link = document.createElement("a");
                          link.href = url;
                          link.download = d.fileName;
                          link.click();
                          URL.revokeObjectURL(url);
                        }
                        toast("Downloading document...");
                      }}
                      className="grid h-8 w-8 place-items-center rounded-lg text-muted-foreground hover:bg-primary/10 hover:text-primary transition" 
                      title="Download"
                    >
                      <Download className="h-4 w-4" />
                    </button>
                  </td>
                </tr>
              ))}
              {visibleDocs.length === 0 && (
                <tr><td colSpan={6} className="px-4 py-16 text-center text-sm text-muted-foreground">No documents uploaded yet.</td></tr>
              )}
            </tbody>
          </table>
        </div>
      </Card>
      {open && (
        <ModalShell title="Upload Document" onClose={() => setOpen(false)}>
          <div className="space-y-4">
            {!scopeToSelf && (
              <Field label="Employee">
                <select value={selectedEmp} onChange={e => setSelectedEmp(e.target.value)} className={inputCls}>
                  {employees.map(e => <option key={e.id}>{e.name}</option>)}
                </select>
              </Field>
            )}
            <Field label="Document Type">
              <select value={docType} onChange={e => setDocType(e.target.value)} className={inputCls}>
                {["Aadhaar Card","PAN Card","Passport","Degree Certificate","Experience Certificate","Bank Passbook","Offer Letter Copy","Other"].map(t => <option key={t}>{t}</option>)}
              </select>
            </Field>
            <Field label="File Name (optional)">
              <input value={fileName} onChange={e => setFileName(e.target.value)} className={inputCls} placeholder="e.g. aadhaar_john.pdf" />
            </Field>
            
            <div className="space-y-2">
              <span className="text-xs font-semibold text-muted-foreground">Select File</span>
              <div className="flex flex-col items-center justify-center border-2 border-dashed border-border rounded-xl p-6 bg-muted/20 relative">
                <input type="file" onChange={handleFileChange} className="absolute inset-0 opacity-0 cursor-pointer" />
                <FolderOpen className="h-8 w-8 text-muted-foreground mb-2" />
                <span className="text-sm font-semibold text-primary">Choose File to Upload</span>
                <span className="text-xs text-muted-foreground mt-1">PDF, JPG, PNG up to 10MB</span>
                {fileObj && (
                  <div className="mt-4 p-2 bg-card rounded border border-border text-xs w-full max-w-[280px] text-center truncate">
                    <span className="font-semibold">{fileObj.name}</span> ({fileObj.size})
                  </div>
                )}
              </div>
            </div>
          </div>
          <div className="mt-6 flex justify-end gap-2"><Btn variant="outline" onClick={() => setOpen(false)}>Cancel</Btn><Btn onClick={upload}><Upload className="h-4 w-4" /> Upload</Btn></div>
        </ModalShell>
      )}
    </div>
  );
}

function letterBodyFor(doc: DocRecord, company: string): string {
  if (doc.type === "Offer") {
    return `Dear ${doc.employee},\n\nWe are pleased to offer you the position of <b>${doc.designation}</b> at ${company}. Based on our discussions and your qualifications, we believe you will be a valuable addition to our team.\n\n<b>Compensation:</b> Your annual gross salary will be <b>₹${doc.salary.toLocaleString()}</b>, payable monthly, subject to statutory deductions.\n\n<b>Reporting date:</b> ${doc.issued}. Please confirm your acceptance by countersigning this offer.\n\nWe look forward to welcoming you.`;
  }
  if (doc.type === "Appointment") {
    return `Dear ${doc.employee},\n\nFurther to your acceptance of our offer, we are delighted to formally appoint you as <b>${doc.designation}</b> at ${company}, effective <b>${doc.issued}</b>.\n\nYour total annual compensation of <b>₹${doc.salary.toLocaleString()}</b> will be governed by the company's HR policies. This appointment is subject to satisfactory completion of the probation period and applicable statutory verifications.\n\nWelcome aboard.`;
  }
  return `To Whom It May Concern,\n\nThis is to certify that <b>${doc.employee}</b> was employed with ${company} as <b>${doc.designation}</b>. During the tenure, ${doc.employee.split(" ")[0]} demonstrated professionalism, integrity, and strong contribution to the team.\n\nWe wish ${doc.employee.split(" ")[0]} every success in future endeavors.\n\nIssued on: ${doc.issued}`;
}

  const html = `<!doctype html><html><head><meta charset="utf-8"><title>${doc.type} - ${doc.employee}</title>
<style>
body{font-family:Georgia,serif;color:#1f2937;margin:0;padding:60px;background:#fff;max-width:800px;line-height:1.7}
.header{display:flex;justify-content:space-between;align-items:center;border-bottom:2px solid #7c3aed;padding-bottom:24px;margin-bottom:32px}
.brand{display:flex;align-items:center;gap:12px}
.logo{width:48px;height:48px;background:linear-gradient(135deg,#7c3aed,#c026d3);border-radius:12px;display:grid;place-items:center;color:#fff;font-weight:800;font-family:sans-serif}
h2{margin:0;font-size:22px;font-family:sans-serif}
.meta{font-size:12px;color:#6b7280;font-family:sans-serif}

</style></head><body>

<script>window.onload = () => { window.print(); }</script>
</body></html>`;
  printHTML(html);
}

                <div className="font-bold">{company}</div>
                <div className="text-xs text-muted-foreground">221B Baker St, London</div>
              </div>
            </div>
            <div className="text-right text-xs text-muted-foreground">
              <div>Ref: {doc.id}</div><div>Date: {doc.issued}</div>
            </div>
          </div>
          <div className="text-center uppercase tracking-widest text-primary font-bold my-6">{doc.type} Letter</div>
          <div className="text-sm leading-relaxed whitespace-pre-wrap" style={{ fontFamily: "Georgia, serif" }} dangerouslySetInnerHTML={{ __html: body.replace(/\n/g, "<br/>") }} />
          <div className="mt-10">
            <div>Sincerely,</div>
            <div className="mt-8 font-bold">Deepak Bhagat</div>
            <div className="text-xs text-muted-foreground">HR Manager - ${company}</div>
          </div>
        </div>
      </div>
      <div className="mt-6 flex justify-end gap-2">
        <Btn variant="outline" onClick={onClose}>Close</Btn>
        <Btn onClick={() => generateDocumentPDF(doc, company)}><Printer className="h-4 w-4" /> Print / Save PDF</Btn>
      </div>
    </ModalShell>
  );
}
